// =========================================================
// ===== помощники ассертов (gosub-подпрограммы) ===========
// =========================================================

:assert_num
if #t_num_actual = #t_num_expected
  log PASS $t_name
  set #pass #pass + 1
else
  log FAIL $t_name input=$t_input actual=#t_num_actual expected=#t_num_expected
  set #fail #fail + 1
end_if
return


:assert_str
// Точное сравнение строк без size(): size("") падает (см. BUGS.md),
// поэтому равенство проверяем двойным posex (A префикс B и B префикс A).
set #t_p1 posex($t_str_expected $t_str_actual)
set #t_p2 posex($t_str_actual $t_str_expected)
if (#t_p1 = 1) and (#t_p2 = 1)
  log PASS $t_name
  set #pass #pass + 1
else
  log FAIL $t_name input=$t_input actual=$t_str_actual expected=$t_str_expected
  set #fail #fail + 1
end_if
return


:test_summary
log TEST RESULT PASS=#pass FAIL=#fail
if #fail = 0
  log ALL TESTS PASSED
else
  log SOME TESTS FAILED
end_if
return

// =========================================================
// ===== 14a. операторы сравнения строк ====================
// =========================================================
// Семантика из CheckCompare (uScanThread.pas):
//   = == <> < > : регистронезависимо; числовая ветка если оба операнда Int64.
//   <= >=       : вхождение подстроки (<=: правый в левом; >=: левый в правом).
//   <- ->       : вхождение подстроки без числовой ветки.
// ВАЖНО: EvalScriptExpr СНИМАЕТ обрамляющие кавычки строкового литерала
// (uScanThread.pas:13411-13421), поэтому "5" и 5 неразличимы для сравнения:
// "5" = "05" идёт по числовой ветке (true), "10" <= "0" -- числовое 10<=0
// (false), а НЕ вхождение "0" в "10".

:test_cmp_eq_same_true
set $t_name test_cmp_eq_same_true
set $t_input абв равно абв
if "абв" = "абв"
  set #t_num_actual 1
else
  set #t_num_actual 0
end_if
set #t_num_expected 1
gosub assert_num
return

:test_cmp_eq_ne_false
set $t_name test_cmp_eq_ne_false
set $t_input абв равно абвг
