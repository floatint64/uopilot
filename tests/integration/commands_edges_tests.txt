// Безопасный edge-кейс, который генератор не покрывает:
// `set #r` без значения создаёт переменную и обнуляет её. wait 0/1 покрыт
// сгенерированным flow_tests.txt, delete_array — arrays_tests.txt.
// Кодировка файла: UTF-8.

log clear
log mode noTime
set linedelay 0
set #pass 0
set #fail 0

gosub case_set_without_value
gosub test_summary
exit


// ===== помощники ассертов =====

:assert_num
if #t_num_actual = #t_num_expected
  log PASS $t_name
  set #pass #pass + 1
else
  log FAIL $t_name input=$t_input actual=#t_num_actual expected=#t_num_expected
  set #fail #fail + 1
end_if
return


:test_summary
log TEST RESULT PASS=#pass FAIL=#fail
if #fail = 0
  log ALL TESTS PASSED
else
  log SOME TESTS FAILED
end_if
return


// ===== кейсы =====

:case_set_without_value
set $t_name set_without_value
set $t_input set #r after 7
set #r 7
set #r
set #t_num_actual #r
set #t_num_expected 0
gosub assert_num
return
