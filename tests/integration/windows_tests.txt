// Интеграционные тесты команд работы с окнами:
// findwindow / getwindowtext / setwindowtext / getselectedtext / setselectedtext
// Кодировка файла: UTF-8.
// Каркас на gosub/return: у call/proc изолированная область видимости переменных,
// поэтому подпрограммы-ассерты и счётчики живут в одном потоке через gosub.
// Подопытное окно -- классический notepad (C:\Windows\System32\notepad.exe).
// Порядок важен: B1 ставит уникальный заголовок ДО findwindow-кейсов (A2-A5),
// а B6 (стирание заголовка) выполняется ПОСЛЕ всех A-кейсов.
// C-кейсы (selected) зависят от фокуса; допускается флаки (см. README).
// Выделение всего текста делаем через {home} + ~{end} (Shift+End), а НЕ через
// Ctrl+A: глобальный хоткей UoPilot "Set work window" (Ctrl+A) перехватывает
// Ctrl+A и не даёт notepad выделить текст. Возможно поможет опция "прозрачные" хоткеи.

log clear
log mode noTime

// ===== счётчики =====
set #pass 0
set #fail 0

// ===== сетап: запуск notepad и поиск по классу =====
// На медленных машинах классический notepad через WOW64-заглушку создаёт
// окно с классом "Notepad" не мгновенно, поэтому ждём с повторными попытками.
exec C:\Windows\System32\notepad.exe
wait 2000
set #h findwindow (Notepad)
if #h = 0
  wait 1000
  set #h findwindow (Notepad)
end_if
if #h = 0
  wait 1500
  set #h findwindow (Notepad)
end_if
gosub test_setup_found

// ===== B1: уникальный заголовок ДО findwindow-кейсов =====
gosub test_swt_ok

// ===== A: findwindow =====
gosub test_fw_exact_cyr
gosub test_fw_prefix_cyr
gosub test_fw_substr_lat
gosub test_fw_casefold
gosub test_fw_notfound
gosub test_fw_empty

// ===== B: getwindowtext / setwindowtext =====
gosub test_gwt_roundtrip
gosub test_gwt_invalid
gosub test_swt_invalid
gosub test_swt_null
gosub test_swt_empty

// ===== C: сетап selected-кейсов (фокус на Edit) =====
set #r setwindowtext (#h UoPilotSelTest)
showwindow #h
wait 400
set #ed getwindow (#h child)
gosub test_sel_editfound

gosub test_set_sel_write
gosub test_sel_write_cross
gosub test_get_sel_empty
gosub test_get_sel_full
gosub test_set_sel_replace
gosub test_get_sel_replaced

// ===== D: очистка =====
set #r setwindowtext (#h UoPilotWndTestDone)
terminate UoPilotWndTestDone

gosub test_summary
exit


// =========================================================
// ===== помощники ассертов (gosub-подпрограммы) ===========
// =========================================================

:assert_num
if #t_num_actual = #t_num_expected
  log PASS $t_name
  set #pass #pass + 1
else
  log FAIL $t_name input=$t_input actual=#t_num_actual expected=#t_num_expected
  set #fail #fail + 1
end_if
return


:assert_str
// Точное сравнение строк без size(): size("") падает (см. BUGS.md),
// поэтому равенство проверяем двойным posex (A префикс B и B префикс A).
set #t_p1 posex($t_str_expected $t_str_actual)
set #t_p2 posex($t_str_actual $t_str_expected)
if (#t_p1 = 1) and (#t_p2 = 1)
  log PASS $t_name
  set #pass #pass + 1
else
  log FAIL $t_name input=$t_input actual=$t_str_actual expected=$t_str_expected
  set #fail #fail + 1
end_if
return


:test_summary
log TEST RESULT PASS=#pass FAIL=#fail
if #fail = 0
  log ALL TESTS PASSED
else
  log SOME TESTS FAILED
end_if
return
