// Интеграционные тесты команд буфера обмена: set clipboard / get clipboard
// Кодировка файла: UTF-8.
// Каркас на gosub/return: у call/proc изолированная область видимости переменных,
// поэтому подпрограммы-ассерты и счётчики живут в одном потоке через gosub.

log clear
log mode noTime

set #pass 0
set #fail 0

gosub test_set_lat_lower
gosub test_set_lat_upper
gosub test_set_cyr_lower
gosub test_set_cyr_upper
gosub test_set_mixed
gosub test_set_yo
gosub test_set_space
gosub test_set_digits
gosub test_set_from_var
gosub test_errorlevel_set
gosub test_errorlevel_get
gosub test_summary
exit


// ===== helpers =====

:assert_num
if #t_num_actual = #t_num_expected
  log PASS $t_name
  set #pass #pass + 1
else
  log FAIL $t_name input=$t_input actual=#t_num_actual expected=#t_num_expected
  set #fail #fail + 1
end_if
return


:assert_str
set #t_p1 posex($t_str_expected $t_str_actual)
set #t_p2 posex($t_str_actual $t_str_expected)
if (#t_p1 = 1) and (#t_p2 = 1)
  log PASS $t_name
  set #pass #pass + 1
else
  log FAIL $t_name input=$t_input actual=$t_str_actual expected=$t_str_expected
  set #fail #fail + 1
end_if
return


:test_summary
log TEST RESULT PASS=#pass FAIL=#fail
if #fail = 0
  log ALL TESTS PASSED
else
  log SOME TESTS FAILED
end_if
return


// ===== A. round-trip set/get =====
